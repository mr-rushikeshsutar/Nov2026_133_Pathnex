#jenkins - Simulated Rollback

pipeline {

agent any 

environment {
INSTITUTE_NAME = "PATHNEX"
}
stages{
stage('build'){
steps{
script{
try{
sh 'echo "deploying to $INSTITUTE_NAME"'
sh 'exit 1' // simulate failuer
}
catch (Exception e) {
 echo "deployment failed rolling back $INSTITUTE_NAME"
}
}
}
}
}
}


#gitlab - Simulated Rollback


stages:
 - deploy

variables:
 INSTITUTE_NAME: "PATHNEX"

deploy:
 stage: deploy
 image: alpine:latest
 script:
  - echo "deploying $INSTITUTE_NAME"
  - exit 1
 after_script:
  - echo "Deployment failed, rolling back $INSTITUTE_NAME"
