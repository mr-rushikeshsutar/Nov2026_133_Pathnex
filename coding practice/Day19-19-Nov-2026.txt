#ansible - Deploy App Using Git

---
- name: deploy from git 
  hosts: all
  become: yes
 
  tasks:
   - name: install git 
      yum:
       name: git
       state: present

   - name: clone repo
     git:
      repo: "https://github.com/pathnex/app.git"
      dest: "/opt/pathnex-app"
      version: main


#terraform - RDS Skeleton (MySQL)


resource "aws_db_instance" "pathnex-db"{

allocated_storage = 20
engine = "mysql"
instance_class = "db.t3.micro"
username = "admin"
password = "pathnex"
skip_final_snapshot = true
}


#kubernetes - Deployment Rolling Update

apiVersion: apps/v1
kind: Deployment
metadata:
 name: pathnex-rollout

spec:
 replicas: 3
 strategy:
  type: RollingUpdate
   rollingUpdate: 
     maxSurge: 2
     maxUnavailable: 1
 
 selector:
  matchLabels:
   app: rollout
 
 template:
  metadata:
   labels:
    app: rollout

  spec:
   containers:
    - name: app
      image: nginx
      ports:
      - containerPort: 80

#shell-script - Check Internet Connectivity

fi

#jenkins - Approvals


pipeline {

agent any 

environment {
INSTITUTE_NAME = "PATHNEX"
}

stages {
stage ('deploy'){
steps{
input message: "Approve deployment for $INSTITUTE_NAME?"
sh 'echo "deployment approved for $INSTITUTE_NAME"'
}
}

}
}
