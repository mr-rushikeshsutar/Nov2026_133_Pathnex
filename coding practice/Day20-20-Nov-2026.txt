#!/bin/bash
service="nginx"
if systemctl $service  is-active --quiet ; then
echo "$service is running"
else
systemctl start $service 

echo "restarting $service"

#jenkins - Secrets & Credentials

pipeline {

agent any 

environment {
INSTITUTE_NAME = "PATHNEX"
}

stages{

stage('checkout'){
steps{

git branch 'main', url: 'https://github.com/Pathnex/sample-java-app.git'

}
}

stage('use-secret') {
steps{
withCredentials([string(credentialsID:'PATHNEX_API_KEY', variable:'API_KEY')]){
sh 'echo "using secrete $API_KEY for $INSTITUTE_NAME"'
}
}
}
}

}
