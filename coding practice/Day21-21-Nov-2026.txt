#ansible - nginx, Start enable service

---
- name: install nginx
  hosts: all
  become: yes

  tasks:
  - name: install nginx
    yum:  
     name: nginx
     state: present

  - name: start service
    service:
     name: nginx
     state: started
     enabled: yes

#jenkins - Multi-Environment Pipeline


pipeline {
agent any 
environment {
INSTITUTE_NAME = "PATHNEX" 
}

stages{
stage('deploy to dev') {
steps{
 sh 'echo" deploying $INSTUTUTE_NAME to dev "'
}

}

stage('deploy to stagging') {
steps{
input message : "APPROVE STAGING DEPLOYMENT?"
sh 'echo" deploying $INSTUTUTE_NAME to staging "'
}

}

stage('deploy to prod'){

steps{
input message: "APPROVE $INSTITUTE_NAME TO PROD"
sh 'echo" deploying $INSTUTUTE_NAME to prod "  '
}
}

}
}
