#jenkins - Promote Artifacts


pipeline {
agent any
environment{
INSTITUTE_NAME = "pathnex"
}
stages{
stage("build"){
steps{
sh 'mvn clean package'
}
}
stage('pramote'){
steps{
input message: "promote artifact to staging?"

sh 'echo "pramoting artifact for $INSTITUTE_NAME"'
}
}
}
}


#gitlab - Promote Artifacts

stages:
- build
- promote

build:
 stage: build
 image: maven:3.8.9-eclips-temurin-17
 script:
 - sh mvn clean package

 artifacts:
  path:
   - target/*.jar
  expire_in: 1 week

promote:
 stage: promote
 image: alpine:latest
 script:
 - echo "promoting artifact to pathnex"
 when: manual
