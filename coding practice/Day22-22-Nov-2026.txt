#ansible - Handler on Config Change

---
- name: configure nginx for pathnex
  become: yes
  hosts: all 

  tasks:
  - name: copy nginx config 
    copy:
     dest: /etc/nginx/nginx.conf
     content: |
       user nginx;
       worker_processes auto;
    notify: restart nginx

  handlers:
   - name: restart nginx
     service:
      name: nginx
      state: restarted 


#terraform - Security Group

}

#shell-script - Monitor Disk Usage
#!/bin/bash

#jenkins - Trigger on Tag

pipeline {

agent any 

environment {
INSTITUTE_NAME= "PATHNEX"
TEAM= "DevOps"
ENV= "prod"
}

triggers {
pollSCM('H/15 * * * *')
}

stages{
stage('checkout'){
steps{
git branch 'main', url: 'https://github.com/Pathnex/sample-java-app.git'
}
}


stage('build'){
when {
buildingTag()
}

steps {
sh 'echo "building $INSTITUTE_NAME app for $TEAM in $ENV environment"' 

}

}

}
}
