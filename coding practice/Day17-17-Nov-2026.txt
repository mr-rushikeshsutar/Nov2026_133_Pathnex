#ansible - Create Multiple Users in Loop

---
- name: create multiple role
  hosts: all
  become: yes
  
  vars:
   users:
   - dev1
   - dev2
   - dev3

  tasks:
  - name: create-user
    user:
     name: "{{item}}"
     state: present
    loop: "{{users}}"

#kubernetes - Namespace + Deployment

---
apiVersion : v1
kind: Namespace
metadata:
 name: pathnex

---
apiVersion : apps/v1
kind: Deployment
metadata: 
 name: pathnex-deployment
 namespace: pathnex

spec:
 replicas: 2
 selector:
  matchLabels:
   app: "pathnex"

 template:
  metadata:
   labels:
    app: "pathnex"
  
  spec:
   containers: 
    - name: app
      image: nginx
      ports:
       - containerPort: 80

#jenkins - Simulated Deployment


pipeline {
agent any 

environment {
INSTITUTE_NAME= "PATHNECX"
}

parameters{
choice(
name: 'ENVIRONMENT', 
choices: ['dev', 'stage' ,'prod'],
description: "Deployment Environment"
)
}

stages{
stage('build') {
steps{
sh 'mvn clean package'
}
}

stage('deploy') {
steps{
 sh 'echo "deploying $INSTITUTE_NAME application to $ENVIRONMENT"'
}
}

}
}
